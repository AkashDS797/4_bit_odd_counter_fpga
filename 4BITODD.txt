module odd_counter4bit #(
    parameter integer CLK_FREQ_HZ = 50_000_000
)
(
    input wire clk,
    input wire reset,
    output reg [3:0] led
);

    reg [31:0] clk_count;

    always @(posedge clk)
    begin
        if (reset)
        begin
            clk_count <= 0;
            led <= 4'b0001;
        end

        else if (clk_count == CLK_FREQ_HZ - 1)
        begin
            clk_count <= 0;

            if (led == 4'b1111)
                led <= 4'b0001;
            else
                led <= led + 4'd2;
        end

        else
        begin
            clk_count <= clk_count + 1'b1;
        end
    end

endmodule
